function importFullBackup(e) { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = async function(evt) { try { const parsed = JSON.parse(evt.target.result); let decksToImport = []; if (Array.isArray(parsed)) { decksToImport = parsed; } else if (parsed && Array.isArray(parsed.decks)) { decksToImport = parsed.decks; } else { throw new Error("Structure de fichier non reconnue."); } // S'assure de conserver chaque carte, son statut et ses précisions d'origine window.appData.decks = decksToImport.map(deck => ({ title: deck.title || deck.deckName || "Sans titre", category: deck.category || "Général", cards: (deck.cards || []).map(card => ({ id: card.id || 'c_' + Math.random(), question: card.question || "", answer: card.answer || "", precision: card.precision || "", status: card.status || "unseen" // Conserve précieusement ton avancement ! })) })); if (window.saveToFirestore) { await window.saveToFirestore(); } alert(`Succès ! ${decksToImport.length} deck(s) et toute votre progression ont été restaurés dans le Cloud.`); switchTab('library'); } catch (err) { alert("Erreur lors de l'importation de la sauvegarde : " + err.message); } }; reader.readAsText(file); }
